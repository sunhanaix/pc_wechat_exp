```python
def _clean_wxid(wxid):
    """Strip the suffix after the second underscore.
    'wxid_cpyn7pe119rs21_10e8' -> 'wxid_cpyn7pe119rs21'
    """
    if not wxid or not wxid.startswith('wxid_'):
        return wxid
    parts = wxid.split('_')
    if len(parts) >= 3:
        return '_'.join(parts[:2])
    return wxid
```

```python
def _extract_keys_from_mmkv(db_dir, db_files, salt_to_dbs, print_fn):
    """Extract DB encryption keys from MMKV config files (WeChat 4.1.10.x).
    
    MMKV files in db_storage/MMKV/ are AES-GCM encrypted. The AES key is
    derived via MD5(str(code) + cleaned_wxid).hex()[:16].
    
    Returns key_map dict {salt_hex: enc_key_hex} for all found keys.
    """
    key_map = {}

    # Locate MMKV directory
    mmkv_dir = os.path.join(db_dir, 'MMKV')
    if not os.path.isdir(mmkv_dir):
        print_fn("[MMKV] db_storage/MMKV/ not found — skipping MMKV extraction")
        return key_map

    # Extract wxid from db_dir path
    # e.g. D:\xwechat_files\wxid_cpyn7pe119rs21_10e8\db_storage
    wxid_full = os.path.basename(os.path.dirname(db_dir))
    wxid_clean = _clean_wxid(wxid_full)
    if not wxid_clean or not wxid_clean.startswith('wxid_'):
        print_fn(f"[MMKV] Cannot extract wxid from path: {db_dir}")
        return key_map

    print_fn(f"[MMKV] wxid: {wxid_full} -> cleaned: {wxid_clean}")

    # Enumerate MMKV files, extract codes from filenames
    # Pattern: f<hex>tinfo.mmkv  ->  code = int(hex, 16)
    # Also:   storage_common     ->  no code (use default)
    import re as _re
    _CODE_RE = _re.compile(r'^f([0-9a-fA-F]+)tinfo\.mmkv$')

    codes = {}  # code_or_label -> mmkv_filepath
    try:
        for fname in os.listdir(mmkv_dir):
            if fname.endswith('.crc'):
                continue
            m = _CODE_RE.match(fname)
            if m:
                code = int(m.group(1), 16)
                codes[code] = os.path.join(mmkv_dir, fname)
            elif fname.endswith('.mmkv'):
                # storage_common, brandprivatemsg@hardcode, etc.
                # Use None as code (wxid-only fallback)
                codes[fname] = os.path.join(mmkv_dir, fname)
    except OSError as e:
        print_fn(f"[MMKV] Cannot list MMKV directory: {e}")
        return key_map

    if not codes:
        print_fn("[MMKV] No .mmkv files found")
        return key_map

    print_fn(f"[MMKV] Found {len(codes)} MMKV file(s)")

    # Build lookup: rel_path -> (salt_hex, page1)
    path_to_info = {}
    for rel, path, sz, salt_hex, page1 in db_files:
        path_to_info[rel] = (salt_hex, page1)
        # Also try with forward slashes
        path_to_info[rel.replace('\\', '/')] = (salt_hex, page1)

    # Try pycryptodome import (already a project dependency)
    try:
        from Crypto.Cipher import AES
    except ImportError:
        print_fn("[MMKV] pycryptodome not available — skipping MMKV extraction")
        return key_map

    # Try each code
    for code_or_label, filepath in codes.items():
        if not key_map and len(key_map) >= len(salt_to_dbs):
            break

        # Derive AES key
        if isinstance(code_or_label, int):
            aes_key_str = hashlib.md5(
                f"{code_or_label}{wxid_clean}".encode()
            ).hexdigest()[:16]
        else:
            # No numeric code: try wxid-only
            aes_key_str = hashlib.md5(
                wxid_clean.encode()
            ).hexdigest()[:16]
        aes_key = aes_key_str.encode('ascii')

        label = code_or_label if isinstance(code_or_label, int) else f"'{code_or_label}'"
        print_fn(f"[MMKV] Trying code={label} key={aes_key_str}...")

        # Read and decrypt MMKV file
        try:
            with open(filepath, 'rb') as f:
                raw = f.read()
        except (IOError, PermissionError) as e:
            print_fn(f"[MMKV] Cannot read {filepath}: {e}")
            continue

        if len(raw) < 24:
            continue  # Too small: 4B size + 16B IV + min ciphertext

        total_size = struct.unpack('<I', raw[:4])[0]
        if total_size + 4 != len(raw):
            # Size mismatch — try len(raw) - 4 as size (some MMKV variants)
            total_size = len(raw) - 4

        iv = raw[4:20]
        if total_size < 17:
            continue  # Need at least 1 byte ciphertext + 16 byte auth tag

        ciphertext = raw[20:20 + total_size - 16]
        auth_tag = raw[20 + total_size - 16:20 + total_size]

        try:
            cipher = AES.new(aes_key, AES.MODE_GCM, nonce=iv)
            plaintext = cipher.decrypt_and_verify(ciphertext, auth_tag)
        except (ValueError, KeyError) as e:
            # Auth failure or invalid key — wrong code
            print_fn(f"[MMKV] GCM auth failed for code={label}: {e}")
            continue
        except Exception as e:
            print_fn(f"[MMKV] Decryption error for code={label}: {e}")
            continue

        print_fn(f"[MMKV] Decrypted {len(plaintext)} bytes from {os.path.basename(filepath)}")

        # Parse plaintext: search for DB rel_path -> 64-char hex value
        _HEX64_RE = _re.compile(b'([0-9a-fA-F]{64})')

        for rel, (salt_hex, page1) in path_to_info.items():
            if salt_hex in key_map:
                continue

            # Try both path separators
            for sep in [b'\\', b'/']:
                path_bytes = rel.replace('\\', sep.decode()).encode()
                pos = plaintext.find(path_bytes)
                while pos >= 0:
                    # Search nearby bytes for 64-char hex key
                    nearby_start = pos + len(path_bytes)
                    nearby_end = min(len(plaintext), nearby_start + 512)
                    nearby = plaintext[nearby_start:nearby_end]

                    for hex_match in _HEX64_RE.finditer(nearby):
                        candidate_hex = hex_match.group(0).decode()
                        try:
                            candidate_key = bytes.fromhex(candidate_hex)
                        except ValueError:
                            continue
                        if verify_enc_key(candidate_key, page1):
                            key_map[salt_hex] = candidate_hex
                            print_fn(f"\n  [MMKV-FOUND] salt={salt_hex}")
                            print_fn(f"    enc_key={candidate_hex}")
                            print_fn(f"    DB: {rel}")
                            break

                    if salt_hex in key_map:
                        break
                    pos = plaintext.find(path_bytes, pos + 1)

                if salt_hex in key_map:
                    break

    print_fn(f"[MMKV] Extracted {len(key_map)}/{len(salt_to_dbs)} keys from MMKV")
    return key_map
```

```python
"""One-off validation: test MMKV-based key extraction against known keys."""
import json
import os
import sys
sys.path.insert(0, os.path.join(os.path.dirname(__file__), '..', 'src'))

from engine.utils import find_wechat_data_dir
from key_scan import collect_db_files, _clean_wxid, _extract_keys_from_mmkv

def main():
    db_dir = find_wechat_data_dir()
    if not db_dir:
        print("[FAIL] Cannot find WeChat data directory")
        return

    db_files, salt_to_dbs = collect_db_files(db_dir)
    print(f"Found {len(db_files)} DBs, {len(salt_to_dbs)} salts")

    key_map = _extract_keys_from_mmkv(db_dir, db_files, salt_to_dbs, print)
    print(f"\nMMKV extraction result: {len(key_map)}/{len(salt_to_dbs)} keys")

    # Compare against known keys
    config_path = os.path.join(os.path.dirname(__file__), '..', '.wechat_exp_config.json')
    if os.path.exists(config_path):
        with open(config_path, 'r', encoding='utf-8') as f:
            config = json.load(f)
        known = config.get('db_keys', {})
        match = 0
        mismatch = 0
        for rel, key_hex in known.items():
            for salt_hex, (_, page1) in [(s, p) for r, p, sz, s, p1 in db_files if r == rel]:
                if salt_hex in key_map:
                    if key_map[salt_hex] == key_hex:
                        match += 1
                    else:
                        mismatch += 1
                        print(f"  MISMATCH: {rel}")
                    break
        print(f"Match: {match}, Mismatch: {mismatch}, Missing: {len(known) - match - mismatch}")
        if match == len(known):
            print("[PASS] All known keys matched!")
        else:
            print(f"[PARTIAL] {match}/{len(known)} matched")

if __name__ == '__main__':
    main()
```
